\section{方法}

\begin{frame}{密度不确定性估计：替代人为 UQ 估计器}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{原接口保留}
\[
\text{EVAViT 特征} \rightarrow (\text{分数}, \text{表征向量})
\]
\vspace{0.3em}
\textbf{含义改变}
\begin{itemize}
    \item 分数：偏离正常视觉特征流形的程度

\begin{frame}{当前主线：大模型前视觉适配器}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\[
q' = q + s_{\text{UQ}}\cdot
\mathrm{up}(\mathrm{GELU}(\mathrm{down}(\mathrm{LN}(q))))
\]
\vspace{0.2em}
\begin{itemize}
    \item 作用在 大模型前视觉查询向量
    \item 不直接接触 VAE/轨迹解码器
    \item up 零初始化，初始严格 恒等
    \item 可做 正确/打乱/无注入 因果对照
\end{itemize}
\end{column}

\begin{frame}{监督：配对扰动}
\large
\[
d(\text{受损+正确 UQ}, \text{正常参考})
<
d(\text{受损+打乱 UQ}, \text{正常参考})
\]
\vspace{0.8em}
\normalsize
\begin{itemize}
    \item 不手工规定“高 UQ 就刹车”
    \item 正常/受损 是同一场景，几何和专家轨迹一致
    \item 当前扰动：单相机丢失强度 1
    \item 目标：受损视觉下，正确 UQ 帮助恢复接近 正常输入的规划表示
\end{itemize}
\end{frame}

\section{结论与计划}

\begin{frame}{当前能说什么，不能说什么}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{可以说}
\begin{itemize}
    \item 密度不确定性估计比手工 UQ 估计器 更有理论依据
    \item 大模型能读取 UQ 词元的可靠性语义
    \item 大模型前视觉适配器在路线均衡受损输入评估上平均有效
    \item 正确 UQ 优于打乱 UQ
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{暂时不能说}
\begin{itemize}
    \item 显式 UQ 词元已经改善规划
    \item 高 UQ 场景收益一定更大
    \item 适配器已经稳定泛化
    \item 正常场景完全无退化
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{下一阶段最短路线}
\begin{table}
\centering
\small
\begin{tabular}{p{3.2cm}p{5.0cm}p{3.0cm}}
\toprule
\textbf{任务} & \textbf{目标} & \textbf{通过标准} \\
\midrule
路线均衡训练 & 500--1000 配对样本，仅训练适配器 & 正确 UQ 稳定优于打乱 UQ \\
正常输入保持 & 加入恒等/保持损失 & 正常输入 ADE/FDE 退化 $<3\%$ \\
表征向量条件 & 分数和密度表征向量同时注入 & 分层结果更可解释 \\
路线分层 & 按路线/天气/受损强度拆分 & 明确收益和失败边界 \\
\bottomrule
\end{tabular}
\end{table}
\end{frame}

\begin{frame}{总结}
\large
\begin{enumerate}
    \item 0521 证明了“需要 UQ”，但旧方案说服力不足。
    \item 本阶段用密度不确定性估计替换手工 UQ 估计器，保留分数与表征向量接口。
    \item 显式 UQ 词元给出重要负结果：读懂语义不等于改善规划。
    \item 大模型前视觉适配器是当前主线：不直接改轨迹，而是条件化大模型前的视觉证据。
    \item 路线均衡实验已有平均收益，但下一步要解决稳定性和 正常输入 FDE。
\end{enumerate}
\end{frame}
